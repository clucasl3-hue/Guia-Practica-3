# 1.1 Modificar Persona (Clase Padre)
class Persona(ModeloBase):
    def __init__(self, id_registro, nombre, apellido, email, edad=0):
        super().__init__(id_registro)
        self.nombre = nombre
        self.apellido = apellido
        self.email = email
        self.edad = int(edad)

    def a_diccionario(self):
        datos = super().a_diccionario()
        datos["nombre"] = self.nombre
        datos["apellido"] = self.apellido
        datos["email"] = self.email
        datos["edad"] = self.edad
        return datos

# 1.2 Modificar Estudiante (Clase Hija)
class Estudiante(Persona):
    def __init__(self, id_registro, nombre, apellido, email, carnet="", carrera="", edad=0):
        super().__init__(id_registro, nombre, apellido, email, edad)
        self.carnet = carnet
        self.carrera = carrera

    def etiqueta(self):
        return f"{super().etiqueta()} · Edad: {self.edad} años · {self.carnet} · {self.carrera}"

IDEA 2: Regla de Negocio en Matrícula Estados Múltiples y Becas

Escenario: Definir 4 estados académicos según la nota (Aprobado Becado, Aprobado, Recuperación, Reprobado).

Paso 1: models.py (Lógica en Matricula) 
class Matricula(ModeloBase):
    def __init__(self, id_registro, estudiante_id, materia_id, periodo, nota=0.0):
        super().__init__(id_registro)
        self.estudiante_id = int(estudiante_id)
        self.materia_id = int(materia_id)
        self.periodo = periodo
        self.nota = float(nota)

    def estado(self):
        if self.nota >= 18.0:
            return "Aprobado (Becado)"
        elif self.nota >= 14.0:
            return "Aprobado"
        elif self.nota >= 10.0:
            return "Recuperación"
        else:
            return "Reprobado"

    def etiqueta(self):
        return f"Matrícula #{self.id} · Período: {self.periodo} · Nota: {self.nota} · Estado: {self.estado()}"
Paso 2: main.py (Reporte Detallado de Rendimiento) 
def resumen_rendimiento():
    imprimir_titulo("RESUMEN GENERAL DE RENDIMIENTO")

class EstudianteController(ControllerBase):

    def obtener_por_carrera(self, carrera_buscada):
        """Devuelve los estudiantes pertenecientes a la carrera ingresada."""

IDEA 4: Modificación de Atributo con Validación Créditos en Materia

Escenario: Agregar el campo creditos a Materia asegurando que solo acepte enteros mayores a cero.

Paso 1: models.py (Materia) 
class Materia(ModeloBase):
    def __init__(self, id_registro, nombre, codigo, creditos=3):
        super().__init__(id_registro)
        self.nombre = nombre
        self.codigo = codigo
        self.creditos = int(creditos)

    def a_diccionario(self):
        datos = super().a_diccionario()
        datos["nombre"] = self.nombre
        datos["codigo"] = self.codigo
        datos["creditos"] = self.creditos
        return datos

    def etiqueta(self):
        return f"{self.codigo} - {self.nombre} · {self.creditos} Créditos"
Paso 2: views.py (MateriaController) 
class MateriaController(ControllerBase):
    def validar(self, datos, id_actual=None):
        # Validaciones de nombre y código...
        
        # Validar créditos positivos
        if "creditos" in datos:
            try:
                val = int(datos["creditos"])
                if val <= 0:
                    return "Los créditos deben ser mayores a 0"
            except ValueError:
                return "Los créditos deben ser un número entero válido"
        return None
Paso 3: main.py

Puntos Críticos a Recordar en Proyecto Académico:
  • Manejo de Heredados: Si agregas `edad` o `telefono` a `Estudiante`, pásalo por `super().__init__()` hacia `Persona`.
  • Diccionario de Opciones: Si agregas un filtro, recuerda vincular la función al diccionario `MENU_ESTUDIANTES`, `MENU_MATERIAS` o `MENU_MATRICULAS`.
  • Valores por Defecto: Asigna valores iniciales (`creditos=3`, `edad=0`) en las firmas de `__init__` para compatibilidad con registros antiguos del JSON.
