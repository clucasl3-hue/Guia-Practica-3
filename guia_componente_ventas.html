class Persona(ModeloBase):
    def __init__(self, id_registro, nombre, apellido, email, telefono=""):
        super().__init__(id_registro)
        self.nombre = nombre
        self.apellido = apellido
        self.email = email
        self.telefono = telefono

    def a_diccionario(self):
        datos = super().a_diccionario()
        # ... nombre, apellido, email ...
        datos["telefono"] = self.telefono
        return datos

# 1.2 Pasarlo en el Hijo (Cliente) mediante super()
class Cliente(Persona):

IDEA 2: Modificación de Regla de Negocio Comercial Descuento Automático

Escenario: Aplicar 10% de descuento automático en compras mayores a $100 y reflejarlo en reportes.

Paso 1: models.py (Cálculo Dinámico en Venta) 
class Venta(ModeloBase):
    def __init__(self, id_registro, cliente_id, producto_id, cantidad, precio_unitario):
        super().__init__(id_registro)
        self.cliente_id = int(cliente_id)
        self.producto_id = int(producto_id)
        self.cantidad = int(cantidad)
        self.precio_unitario = float(precio_unitario)

    def subtotal(self):
        return round(self.cantidad * self.precio_unitario, 2)

    def total(self):
        monto = self.subtotal()
        # Regla de negocio: Si supera los $100, aplica 10% de descuento
        if monto > 100.0:
            return round(monto * 0.90, 2)
        return monto

    def etiqueta(self):
        return f"Venta #{self.id} · Cant: {self.cantidad} · Subtotal: ${self.subtotal():.2f} · Total Final: ${self.total():.2f}"
Paso 2: main.py (Reporte de Facturación Acumulada) 
def resumen_ventas():
    imprimir_titulo("RESUMEN DE FACTURACIÓN CON DESCUENTOS")
    lista_ventas = ventas.leer()

IDEA 3: Comunicación de Controladores / Devolución de Stock Anular Venta

Escenario: Al anular una venta, el sistema debe devolver las unidades vendidas al stock del Producto.

Paso 1: views.py (Interacción entre VentaController y ProductoController) 
class VentaController(ControllerBase):
    def __init__(self, cliente_controller, producto_controller):
        super().__init__("ventas.json", Venta)
        self.cliente_controller = cliente_controller
        self.producto_controller = producto_controller  # Referencia al controlador de Productos

    def anular_venta(self, id_venta):
        venta = self.obtener(id_venta)
        if venta is None:
            return False, f"No existe la venta #{id_venta}"

        # 1. Buscar el producto involucrado
        producto = self.producto_controller.obtener(venta.producto_id)
        if producto:
            # 2. Reintegrar las unidades al stock del Producto
            producto.stock += venta.cantidad
            self.producto_controller.guardar()  # Guardar cambios en productos.json

        # 3. Remover la venta y guardar cambios en ventas.json
        self.lista.remove(venta)
        self.guardar()
        return True, f"Venta #{id_venta} anulada. Stock devuelto (+{venta.cantidad} unids)."
Paso 2: main.py (Llamada desde la Interfaz) 
def anular_venta_menu():
    imprimir_titulo("ANULAR VENTA")

IDEA 4: Nuevo Filtro Avanzado en la Interfaz Filtro por Criterio

Escenario: Crear una opción de menú que liste productos por agotarse (stock ≤ 5) o por Categoría.

Paso 1: views.py (Método de Filtrado en ProductoController) 
class ProductoController(ControllerBase):
    # ... métodos crear, actualizar ...

    def obtener_agotandose(self, limite=5):
        """Devuelve únicamente los productos cuyo stock sea menor o igual al límite."""
        return [p for p in self.lista if p.stock <= limite]

Recordatorios Clave para el Día del Examen:
